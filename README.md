```javascript
const name = "Sandip";

```

## Conditions

```javascript
if (condition) {
    // code
}
```

Ternary:

```javascript
const result =
    condition
        ? "Yes"
        : "No";
```

---

## Loops

```javascript
for (let i = 0; i < 5; i++) {
    console.log(i);
}
```

```javascript
for (const item of items) {
    console.log(item);
}
```

---

## Functions

```javascript
function add(a, b) {
    return a + b;
}
```

Arrow:

```javascript
const add = (a, b) => a + b;
```

## Arrays

```javascript
numbers.map();
numbers.filter();
numbers.find();
numbers.reduce();
numbers.forEach();
numbers.some();
numbers.every();
```

---

## Objects

```javascript
const user = {
    name: "Sandip",
    age: 22
};
```

---

## Destructuring

```javascript
const {
    name,
    age
} = user;
```

```javascript
const [
    first,
    second
] = numbers;
```

---

## Spread

```javascript
const copy = {
    ...user
};
```

```javascript
const combined = [
    ...first,
    ...second
];
```

---

## Rest

```javascript
function sum(...numbers) {
    return numbers;
}
```

---

## Optional Chaining

```javascript
user?.address?.city;
```

---

## Nullish Coalescing

```javascript
const name =
    username ?? "Guest";
```

---

## Promise

```javascript
fetch("/api/users")
    .then(response =>
        response.json()
    )
    .then(data => {
        console.log(data);
    })
    .catch(error => {
        console.error(error);
    });
```

## Async/Await

```javascript
async function getUsers() {
    try {
        const response =
            await fetch("/api/users");

        const data =
            await response.json();

        return data;
    } catch (error) {
        console.error(error);
    }
}
```
